cmake_minimum_required(VERSION 3.22)
project(maplesu_jailbreak_providers LANGUAGES C CXX)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "MapleSU providers currently require Android arm64-v8a")
endif()
add_subdirectory(lease)
option(MAPLESU_BUILD_UPSTREAM_REVIEW "Configure unmodified upstream review targets; never APK payloads" OFF)
option(MAPLESU_BUILD_GHOSTLOCK_ADAPTER "Configure derived Ghostlock downstream; explicit target only" OFF)
option(MAPLESU_BUILD_DIRTYFRAG_ADAPTER "Configure derived DirtyFrag first stage and bootstrap" OFF)
if(MAPLESU_BUILD_UPSTREAM_REVIEW OR MAPLESU_BUILD_GHOSTLOCK_ADAPTER)
    add_subdirectory(ghostlock)
endif()
if(MAPLESU_BUILD_UPSTREAM_REVIEW OR MAPLESU_BUILD_DIRTYFRAG_ADAPTER)
    add_subdirectory(dirtyfrag)
endif()
