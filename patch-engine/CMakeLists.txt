cmake_minimum_required(VERSION 3.22)
project(maplesu_patch_engine LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(upstream-sources.cmake)
add_library(maplesu_patch_core STATIC ${MAPLESU_PATCH_SOURCES} src/patch_safety.cpp src/patch_plan.cpp)
target_include_directories(maplesu_patch_core PUBLIC upstream src
    upstream/3rdparty/capstone-5.0.7/include upstream/3rdparty/asmjit2-src/src)
target_compile_definitions(maplesu_patch_core PRIVATE MAPLESU_LIBRARY NDEBUG _CONSOLE
    CAPSTONE_HAS_ARM64 CAPSTONE_USE_SYS_DYN_MEM ASMJIT_STATIC=)
set_target_properties(maplesu_patch_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_executable(maplesu-kernel-analyze src/analyze_main.cpp)
target_link_libraries(maplesu-kernel-analyze PRIVATE maplesu_patch_core)
add_executable(maplesu-kernel-patch src/patch_main.cpp)
target_link_libraries(maplesu-kernel-patch PRIVATE maplesu_patch_core)
if(ANDROID)
    target_link_options(maplesu-kernel-patch PRIVATE -Wl,-z,max-page-size=16384,-z,common-page-size=16384)
endif()
add_executable(maplesu-native-tests tests/core_tests.cpp)
target_link_libraries(maplesu-native-tests PRIVATE maplesu_patch_core)
